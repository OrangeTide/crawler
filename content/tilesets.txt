# Modular kits, one per tileset. A kit names the baked mesh pieces that
# clad that tileset's sectors, each with its footprint in grid units, so
# the theme G cladding pass places them at native scale over the logical
# grid. A tileset with no kit here renders with the procedural surface
# quads as before.
#
# kit <tileset> <name>
#   <role> <LUMP> <fw> <fh> [<fz>] [@N]
# roles: floor ceil wall pillar; fw fh are the ground footprint in grid
# units, fz the height in units (default 1). An optional "@N" token names
# the resident trim atlas the piece samples (default 0), for a kit whose
# art spans more than one trim sheet.

# Tileset 10: the loafbrr Mausoleum Dungeon kit. This is the dedicated
# modular tileset, past the biome range 0..9, so a map clads only when it
# points at it. Pieces authored at native 1 metre = 1 grid unit, so the
# floor and ceiling cover a 3x3 block and the wall spans 3 units and stands
# 3 tall. See content/rooms/pieces.
# This is a multi-material kit: floors, walls, the ceiling slab and the
# pillar each sample their own trim atlas (@0 floor, @1 wall, @2 ceiling
# concrete, @3 pillar trim). Each role lists the native 3-unit piece and a
# 1-unit base piece; the cladding pass uses the largest that fits a merged
# run, so open areas draw native pieces and ragged edges fall back to base
# tiles at the same texel density, not a stretched native piece.
kit 10 mausoleum
  floor  MAUS_FLOOR       3 3   @0
  floor  MAUS_BASE_FLOOR  1 1   @0
  ceil   MAUS_CEIL        3 3   @2
  ceil   MAUS_BASE_CEIL   1 1   @2
  wall   MAUS_WALL        3 1 3 @1
  wall   MAUS_BASE_WALL   1 1 3 @1
  pillar MAUS_PILLAR      1 1 3 @3

# Tileset 11: the SigilsVault modular cathedral kit, kept at its native grand
# scale. The module is 8 units, so a floor and its vaulted ceiling cover an
# 8x8 block, the wall spans 8 units and stands 16 tall (the cladding scales it
# to each sector's ceiling), and the columns rise about 9. Gothic stone and
# marble from the cathedral trim atlas (trim 11). See content/rooms/pieces.
kit 11 cathedral
  floor  CATH_FLOOR   8 8
  ceil   CATH_CEIL    8 8
  wall   CATH_WALL    8 1 16
  pillar CATH_PILLAR  2 2 9

# Tileset 12: the town temple. Reuses the cathedral meshes (CATH_*) but draws
# them from a warm, candlelit tint of the cathedral trim atlas (trim 12), so
# the shrine reads as holy and lived-in rather than the cold dungeon vaults.
kit 12 temple
  floor  CATH_FLOOR   8 8
  ceil   CATH_CEIL    8 8
  wall   CATH_WALL    8 1 16
  pillar CATH_PILLAR  2 2 9

# Tileset 13: the loafbrr Mines and Cave Set, for the crystal grotto. A
# low-poly organic cave kit on a 4-unit module: rocky ground, cave rock
# walls, and a ceiling slab reused from the ground piece (flipped to face
# down). It is a two-material kit, so floors and the ceiling sample the
# ground atlas (@0) and the walls sample the cave-rock atlas (@1). The
# native pieces are organic (they overhang the 4-unit cell and interlock to
# hide the grid); the 1-unit base pieces are flat fillers for ragged edges.
# No pillar role: the cladding pass does not place pillars yet.
kit 13 grotto
  floor  GROTTO_FLOOR       4 4   @0
  floor  GROTTO_BASE_FLOOR  1 1   @0
  ceil   GROTTO_CEIL        4 4   @0
  ceil   GROTTO_BASE_CEIL   1 1   @0
  wall   GROTTO_WALL        4 1 4 @1
  wall   GROTTO_BASE_WALL   1 1 4 @1

# Tileset 14: a cathedral clerestory, to show the emission channel. It reuses
# the cathedral floor and vault (atlas 0, the cathedral trim) but its walls are
# panes of stained glass from the SigilsVault window mesh. The glass samples its
# own diffuse atlas (@1) and an emission atlas (!2, the kit's window_emission
# map), so the glass glows in its own colours even in a dim nave while the stone
# stays scene-lit. A proper window wall (stone frame plus glass) needs per-piece
# multi-material meshes; this proof uses the glass pane alone as a curtain wall.
kit 14 clerestory
  floor  CATH_FLOOR    8 8
  ceil   CATH_CEIL     8 8
  wall   CATH_GLASS    8 1 16 @1 !2
